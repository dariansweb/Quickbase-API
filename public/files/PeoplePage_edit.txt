    <script>
      const REALM = window.location.hostname;
      const TABLE_DBID = "ENTER YOUR TABLE ID HERE";

      /*
        Paste your newly generated application token here.
        Do not use a user token in browser JavaScript.
      */
      const APP_TOKEN = "";

      const FIELD_IDS = {
        recordId: 3,
        name: 6,
        age: 7,
        favoriteColor: 8,
      };

      function createCell(value) {
        const cell = document.createElement("td");
        cell.textContent = value;
        return cell;
      }

      function createEditCell(recordId, name, age, favoriteColor) {
        const cell = document.createElement("td");

        const button = document.createElement("button");

        button.type = "button";
        button.textContent = "Edit";
        button.className = "edit-button";

        button.addEventListener("click", () => {
          beginEditPerson(recordId, name, age, favoriteColor);
        });

        cell.appendChild(button);

        return cell;
      }

      function createColorCell(colorName) {
        const cell = document.createElement("td");

        const wrapper = document.createElement("span");
        wrapper.className = "color-value";

        const swatch = document.createElement("span");
        swatch.className = "color-swatch";
        swatch.style.backgroundColor = colorName || "transparent";

        const label = document.createElement("span");
        label.textContent = colorName;

        wrapper.appendChild(swatch);
        wrapper.appendChild(label);
        cell.appendChild(wrapper);

        return cell;
      }

      function renderRecords(records) {
        const tableBody = document.getElementById("peopleRows");
        tableBody.replaceChildren();

        for (const record of records) {
          const recordId = getFieldValue(record, FIELD_IDS.recordId);

          const name = getFieldValue(record, FIELD_IDS.name);

          const age = getFieldValue(record, FIELD_IDS.age);

          const favoriteColor = getFieldValue(record, FIELD_IDS.favoriteColor);

          const row = document.createElement("tr");

          row.appendChild(createCell(recordId));
          row.appendChild(createCell(name));
          row.appendChild(createCell(age));
          row.appendChild(createColorCell(favoriteColor));
          row.appendChild(createEditCell(recordId, name, age, favoriteColor));
          tableBody.appendChild(row);
        }
      }

      function beginEditPerson(recordId, name, age, favoriteColor) {
        document.getElementById("editRecordId").value = recordId;

        document.getElementById("editPersonName").value = name;

        document.getElementById("editPersonAge").value = age;

        document.getElementById("editPersonColor").value = favoriteColor;

        const status = document.getElementById("editStatus");

        status.classList.remove("error");

        status.textContent = `Editing Quickbase Record ID ${recordId}.`;
      }

      async function getTemporaryToken() {
        const headers = {
          "QB-Realm-Hostname": REALM,
        };

        if (APP_TOKEN.trim()) {
          headers["QB-App-Token"] = APP_TOKEN;
        }

        const response = await fetch(
          `https://api.quickbase.com/v1/auth/temporary/${TABLE_DBID}`,
          {
            method: "GET",
            credentials: "include",
            headers,
          },
        );

        const data = await response.json();

        if (!response.ok) {
          throw new Error(`HTTP ${response.status}: ${JSON.stringify(data)}`);
        }

        if (!data.temporaryAuthorization) {
          throw new Error("Quickbase did not return temporaryAuthorization.");
        }

        return data.temporaryAuthorization;
      }

      function getFieldValue(record, fieldId) {
        return record[String(fieldId)]?.value ?? "";
      }

      function buildPersonPayload(name, age, favoriteColor) {
        return {
          to: TABLE_DBID,

          data: [
            {
              [FIELD_IDS.name]: {
                value: name,
              },

              [FIELD_IDS.age]: {
                value: Number(age),
              },

              [FIELD_IDS.favoriteColor]: {
                value: favoriteColor,
              },
            },
          ],
        };
      }

      function buildUpdatePersonPayload(recordId, name, age, favoriteColor) {
        return {
          to: TABLE_DBID,

          data: [
            {
              [FIELD_IDS.recordId]: {
                value: Number(recordId),
              },

              [FIELD_IDS.name]: {
                value: name,
              },

              [FIELD_IDS.age]: {
                value: Number(age),
              },

              [FIELD_IDS.favoriteColor]: {
                value: favoriteColor,
              },
            },
          ],
        };
      }

      async function updatePerson(event) {
        event.preventDefault();

        const status = document.getElementById("editStatus");

        const recordId = document.getElementById("editRecordId").value;

        const name = document.getElementById("editPersonName").value.trim();

        const age = document.getElementById("editPersonAge").value;

        const favoriteColor = document
          .getElementById("editPersonColor")
          .value.trim();

        try {
          const temporaryToken = await getTemporaryToken();

          const payload = buildUpdatePersonPayload(
            recordId,
            name,
            age,
            favoriteColor,
          );

          const response = await fetch("https://api.quickbase.com/v1/records", {
            method: "POST",

            headers: {
              "QB-Realm-Hostname": REALM,

              Authorization: `QB-TEMP-TOKEN ${temporaryToken}`,

              "Content-Type": "application/json",
            },

            body: JSON.stringify(payload),
          });

          const data = await response.json();

          console.log("Update record response:", data);

          if (!response.ok) {
            throw new Error(`HTTP ${response.status}: ${JSON.stringify(data)}`);
          }

          const updatedRecordId = data.metadata?.updatedRecordIds?.[0];

          status.classList.remove("error");

          status.textContent = `Quickbase Record ID ${updatedRecordId} was updated successfully.`;

          event.target.reset();

          await loadPeople();
        } catch (error) {
          console.error(error);

          status.classList.add("error");

          status.textContent = `Update failed: ${error.message}`;
        }
      }

      async function addPerson(event) {
        // Stop the HTML form from performing its normal page submission.
        // JavaScript will handle the submission through the Quickbase REST API.

        event.preventDefault();

        const status = document.getElementById("addStatus");

        // Read the values entered into the form.

        const name = document.getElementById("personName").value.trim();
        const age = document.getElementById("personAge").value;
        const favoriteColor = document
          .getElementById("personColor")
          .value.trim();

        try {
          // STEP 1
          // Get temporary REST authorization just like our READ request does.

          const temporaryToken = await getTemporaryToken();

          // STEP 2
          // Build the Quickbase record payload we just tested.

          const payload = buildPersonPayload(name, age, favoriteColor);

          // STEP 3
          // Send the new record data to Quickbase.

          // Notice that this endpoint is different from our READ query:

          // READ:
          // POST /v1/records/query

          // CREATE:
          // POST /v1/records

          const response = await fetch("https://api.quickbase.com/v1/records", {
            method: "POST",

            headers: {
              "QB-Realm-Hostname": REALM,

              Authorization: `QB-TEMP-TOKEN ${temporaryToken}`,

              "Content-Type": "application/json",
            },

            body: JSON.stringify(payload),
          });

          // STEP 4
          // Convert the Quickbase JSON response into JavaScript.

          const data = await response.json();

          // For this checkpoint, inspect exactly what Quickbase returned.

          console.log("Add record response:", data);

          if (!response.ok) {
            throw new Error(`HTTP ${response.status}: ${JSON.stringify(data)}`);
          }

          const createdRecordId = data.metadata?.createdRecordIds?.[0];

          // STEP 5
          // Confirm that Quickbase accepted the new record.

          status.classList.remove("error");

          status.textContent = `${name} was added to Quickbase as Record ID ${createdRecordId}.`;

          event.target.reset();
          await loadPeople();
        } catch (error) {
          console.error(error);

          status.classList.add("error");

          status.textContent = `Add record failed: ${error.message}`;
        }
      }

      async function loadPeople() {
        const startTime = performance.now();

        const status = document.getElementById("status");

        try {
          const temporaryToken = await getTemporaryToken();

          const query = {
            from: TABLE_DBID,
            select: [
              FIELD_IDS.recordId,
              FIELD_IDS.name,
              FIELD_IDS.age,
              FIELD_IDS.favoriteColor,
            ],
            sortBy: [
              {
                fieldId: FIELD_IDS.name,
                order: "ASC",
              },
            ],
          };

          const response = await fetch(
            "https://api.quickbase.com/v1/records/query",
            {
              method: "POST",
              headers: {
                "QB-Realm-Hostname": REALM,
                Authorization: `QB-TEMP-TOKEN ${temporaryToken}`,
                "Content-Type": "application/json",
              },
              body: JSON.stringify(query),
            },
          );

          const data = await response.json();

          if (!response.ok) {
            throw new Error(`HTTP ${response.status}: ${JSON.stringify(data)}`);
          }
          renderRecords(data.data);

          const endTime = performance.now();
          const renderTime = endTime - startTime;

          status.classList.remove("error");
          status.textContent = `REST query returned ${data.data.length} record(s) in ${renderTime.toFixed(2)} ms.`;
        } catch (error) {
          console.error(error);
          status.classList.add("error");
          status.textContent = `REST query failed: ${error.message}`;
        }
      }

      document.addEventListener("DOMContentLoaded", () => {
        document
          .getElementById("addPersonForm")
          .addEventListener("submit", addPerson);

        document
          .getElementById("editPersonForm")
          .addEventListener("submit", updatePerson);

        loadPeople();
      });
    </script>
