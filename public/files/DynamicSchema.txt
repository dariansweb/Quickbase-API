      /*
       * ============================================================
       * QUICKBASE CONTEXT
       * ============================================================
       *
       * The Code Page should discover the app it is running inside.
       * We are not hard-coding an App DBID.
       */

      const REALM = window.location.hostname.includes("quickbase.com")
        ? window.location.hostname
        : "YOUR_REALM.quickbase.com";

      function getAppDbid() {
        const match = window.location.pathname.match(/\/db\/([^/]+)/);

        if (!match) {
          throw new Error(
            "Could not determine the App DBID from the Quickbase URL.",
          );
        }

        return match[1];
      }

      /*
       * Keep the Application Token empty in the shared lesson file.
       */
      const APP_TOKEN = "";

      /*
       * We will populate this only after the page confirms that
       * it is running inside a Quickbase application.
       */
      let APP_DBID = "";

      /*
       * ============================================================
       * TEMPORARY REST AUTHORIZATION
       * ============================================================
       *
       * Same signed-in Code Page authorization pattern used in
       * previous REST lessons.
       */

      async function getTemporaryToken(dbid) {
        const headers = {
          "QB-Realm-Hostname": REALM,
        };

        if (APP_TOKEN.trim()) {
          headers["QB-App-Token"] = APP_TOKEN;
        }

        const response = await fetch(
          `https://api.quickbase.com/v1/auth/temporary/${dbid}`,
          {
            method: "GET",
            credentials: "include",
            headers,
          },
        );

        const data = await response.json();

        if (!response.ok) {
          throw new Error(`HTTP ${response.status}: ${JSON.stringify(data)}`);
        }

        if (!data.temporaryAuthorization) {
          throw new Error("Quickbase did not return temporaryAuthorization.");
        }

        return data.temporaryAuthorization;
      }

      async function inspectLessonSchema() {
        const status = document.getElementById("status");
        const output = document.getElementById("output");

        try {
          APP_DBID = getAppDbid();

          status.classList.remove("success");
          status.textContent = "Inspecting current Quickbase schema...";

          const temporaryToken = await getTemporaryToken(APP_DBID);
          const tablesEndpoint = `https://api.quickbase.com/v1/tables?appId=${APP_DBID}`;

          const tablesResponse = await fetch(tablesEndpoint, {
            method: "GET",
            headers: {
              "QB-Realm-Hostname": REALM,
              Authorization: `QB-TEMP-TOKEN ${temporaryToken}`,
              "Content-Type": "application/json",
            },
          });

          const tables = await tablesResponse.json();

          if (!tablesResponse.ok) {
            throw new Error(
              `HTTP ${tablesResponse.status}: ${JSON.stringify(tables)}`,
            );
          }

          const people = tables.find(
            (table) => table.name.trim().toLowerCase() === "people",
          );

          const departments = tables.find(
            (table) => table.name.trim().toLowerCase() === "departments",
          );

          const tasks = tables.find(
            (table) => table.name.trim().toLowerCase() === "tasks",
          );

          lessonState.peopleTableDbid = people ? people.id : null;
          lessonState.departmentsTableDbid = departments
            ? departments.id
            : null;
          lessonState.tasksTableDbid = tasks ? tasks.id : null;

          lessonState.departmentNameFieldId = null;
          lessonState.departmentLocationFieldId = null;
          lessonState.personNameFieldId = null;
          lessonState.taskNameFieldId = null;
          lessonState.taskDueDateFieldId = null;
          lessonState.taskStatusFieldId = null;
          lessonState.departmentsPeopleRelationshipId = null;
          lessonState.peopleTasksRelationshipId = null;

          let departmentFields = [];
          let peopleFields = [];
          let taskFields = [];
          let peopleRelationships = { relationships: [] };
          let taskRelationships = { relationships: [] };

          if (departments) {
            const departmentToken = await getTemporaryToken(departments.id);

            const fieldsResponse = await fetch(
              `https://api.quickbase.com/v1/fields?tableId=${departments.id}`,
              {
                method: "GET",
                headers: {
                  "QB-Realm-Hostname": REALM,
                  Authorization: `QB-TEMP-TOKEN ${departmentToken}`,
                  "Content-Type": "application/json",
                },
              },
            );

            departmentFields = await fieldsResponse.json();

            if (!fieldsResponse.ok) {
              throw new Error(
                `HTTP ${fieldsResponse.status}: ${JSON.stringify(departmentFields)}`,
              );
            }

            const departmentName = departmentFields.find(
              (field) => field.label === "Department Name" && field.mode === "",
            );

            const location = departmentFields.find(
              (field) => field.label === "Location" && field.mode === "",
            );

            lessonState.departmentNameFieldId = departmentName
              ? departmentName.id
              : null;
            lessonState.departmentLocationFieldId = location
              ? location.id
              : null;
          }

          if (people) {
            const peopleToken = await getTemporaryToken(people.id);

            const peopleFieldsResponse = await fetch(
              `https://api.quickbase.com/v1/fields?tableId=${people.id}`,
              {
                method: "GET",
                headers: {
                  "QB-Realm-Hostname": REALM,
                  Authorization: `QB-TEMP-TOKEN ${peopleToken}`,
                  "Content-Type": "application/json",
                },
              },
            );

            peopleFields = await peopleFieldsResponse.json();

            if (!peopleFieldsResponse.ok) {
              throw new Error(
                `HTTP ${peopleFieldsResponse.status}: ${JSON.stringify(peopleFields)}`,
              );
            }

            const personNameField = peopleFields.find(
              (field) => field.label === "Name" && field.mode === "",
            );

            lessonState.personNameFieldId = personNameField
              ? personNameField.id
              : null;

            const relationshipsResponse = await fetch(
              `https://api.quickbase.com/v1/tables/${people.id}/relationships`,
              {
                method: "GET",
                headers: {
                  "QB-Realm-Hostname": REALM,
                  Authorization: `QB-TEMP-TOKEN ${peopleToken}`,
                  "Content-Type": "application/json",
                },
              },
            );

            peopleRelationships = await relationshipsResponse.json();

            if (!relationshipsResponse.ok) {
              throw new Error(
                `HTTP ${relationshipsResponse.status}: ${JSON.stringify(peopleRelationships)}`,
              );
            }

            if (departments) {
              const relationship = (
                peopleRelationships.relationships || []
              ).find(
                (item) =>
                  item.parentTableId === departments.id &&
                  item.childTableId === people.id,
              );

              lessonState.departmentsPeopleRelationshipId = relationship
                ? relationship.id
                : null;
            }
          }

          if (tasks) {
            const tasksToken = await getTemporaryToken(tasks.id);

            const taskFieldsResponse = await fetch(
              `https://api.quickbase.com/v1/fields?tableId=${tasks.id}`,
              {
                method: "GET",
                headers: {
                  "QB-Realm-Hostname": REALM,
                  Authorization: `QB-TEMP-TOKEN ${tasksToken}`,
                  "Content-Type": "application/json",
                },
              },
            );

            taskFields = await taskFieldsResponse.json();

            if (!taskFieldsResponse.ok) {
              throw new Error(
                `HTTP ${taskFieldsResponse.status}: ${JSON.stringify(taskFields)}`,
              );
            }

            const taskName = taskFields.find(
              (field) => field.label === "Task Name" && field.mode === "",
            );

            const dueDate = taskFields.find(
              (field) => field.label === "Due Date" && field.mode === "",
            );

            const statusField = taskFields.find(
              (field) => field.label === "Status" && field.mode === "",
            );

            lessonState.taskNameFieldId = taskName ? taskName.id : null;
            lessonState.taskDueDateFieldId = dueDate ? dueDate.id : null;
            lessonState.taskStatusFieldId = statusField ? statusField.id : null;

            const taskRelationshipsResponse = await fetch(
              `https://api.quickbase.com/v1/tables/${tasks.id}/relationships`,
              {
                method: "GET",
                headers: {
                  "QB-Realm-Hostname": REALM,
                  Authorization: `QB-TEMP-TOKEN ${tasksToken}`,
                  "Content-Type": "application/json",
                },
              },
            );

            taskRelationships = await taskRelationshipsResponse.json();

            if (!taskRelationshipsResponse.ok) {
              throw new Error(
                `HTTP ${taskRelationshipsResponse.status}: ${JSON.stringify(taskRelationships)}`,
              );
            }

            if (people) {
              const peopleTasksRelationship = (
                taskRelationships.relationships || []
              ).find(
                (item) =>
                  item.parentTableId === people.id &&
                  item.childTableId === tasks.id,
              );

              lessonState.peopleTasksRelationshipId = peopleTasksRelationship
                ? peopleTasksRelationship.id
                : null;
            }
          }

          const state = {
            peopleExists: Boolean(people),
            departmentsExists: Boolean(departments),
            departmentFieldsExist: Boolean(
              lessonState.departmentNameFieldId &&
              lessonState.departmentLocationFieldId,
            ),
            relationshipExists: Boolean(
              lessonState.departmentsPeopleRelationshipId,
            ),
            tasksExists: Boolean(tasks),
            taskFieldsExist: Boolean(
              lessonState.taskNameFieldId &&
              lessonState.taskDueDateFieldId &&
              lessonState.taskStatusFieldId,
            ),
            peopleTasksRelationshipExists: Boolean(
              lessonState.peopleTasksRelationshipId,
            ),
          };

          setButtonStates(state);
          renderLessonProgress(state);

          status.classList.add("success");
          status.textContent =
            "Lesson status rebuilt from the current Quickbase schema.";

          output.textContent = JSON.stringify(
            {
              realm: REALM,
              appDbid: APP_DBID,
              detectedState: state,
              lessonState,
            },
            null,
            2,
          );
        } catch (error) {
          console.error(error);

          status.classList.remove("success");
          status.textContent = `Unable to inspect lesson schema: ${error.message}`;

          output.textContent = JSON.stringify(
            {
              error: error.message,
              lessonState,
            },
            null,
            2,
          );
        }
      }

            description: "Departments used in Lesson 10",
            singleRecordName: "Department",
            pluralRecordName: "Departments",

      /*
       * ============================================================
       * CREATE DEPARTMENT FIELDS
       * ============================================================
       *
       * Now that Quickbase has created the Departments table and
       * returned its Table DBID, we can target that table directly.
       *
       * These are OUR explicit field-creation requests:
       *
       *   Department Name
       *   Location
       *
       * Quickbase's default system fields already exist and were
       * created automatically when the table was created.
       */

      async function createDepartmentFields() {
        const status = document.getElementById("status");
        const output = document.getElementById("output");
        const button = document.getElementById("createDepartmentFieldsButton");

        if (!lessonState.departmentsTableDbid) {
          status.classList.remove("success");
          status.textContent =
            "Create the Departments table before creating its custom fields.";
          return;
        }

        status.classList.remove("success");
        status.textContent = "Creating Department Name and Location...";
        output.textContent = "Loading...";
        button.disabled = true;

        try {
          const tableDbid = lessonState.departmentsTableDbid;
          const temporaryToken = await getTemporaryToken(tableDbid);
          const endpoint = `https://api.quickbase.com/v1/fields?tableId=${tableDbid}`;

          const fieldRequests = [
            {
              label: "Department Name",
              fieldType: "text",
              appearsByDefault: true,
              addToForms: true,
            },
            {
              label: "Location",
              fieldType: "text",
              appearsByDefault: true,
              addToForms: true,
            },
          ];

          const fieldResults = [];

          /*
           * Create the fields one at a time.
           *
           * This is intentional for the lesson:
           * each request produces its own Quickbase response,
           * including the actual Field ID assigned by Quickbase.
           */
          for (const payload of fieldRequests) {
            console.log("Create field endpoint:", endpoint);
            console.log("Create field payload:", payload);

            const response = await fetch(endpoint, {
              method: "POST",
              headers: {
                "QB-Realm-Hostname": REALM,
                Authorization: `QB-TEMP-TOKEN ${temporaryToken}`,
                "Content-Type": "application/json",
              },
              body: JSON.stringify(payload),
            });

            const data = await response.json();

            console.log(`Create field response for ${payload.label}:`, data);

            if (!response.ok) {
              throw new Error(
                `HTTP ${response.status} while creating ${payload.label}: ${JSON.stringify(data)}`,
              );
            }

            /*
             * Capture the Field IDs Quickbase actually assigned.
             * Never assume a Field ID from nextFieldId alone.
             */
            if (payload.label === "Department Name") {
              lessonState.departmentNameFieldId = data.id;
            }

            if (payload.label === "Location") {
              lessonState.departmentLocationFieldId = data.id;
            }

            fieldResults.push({
              request: {
                method: "POST",
                endpoint,
                payload,
              },
              quickbaseResponse: data,
            });
          }

          document.getElementById(
            "createDepartmentsPeopleRelationshipButton",
          ).disabled = false;

          status.classList.add("success");
          status.textContent = `Department fields created — Department Name FID ${lessonState.departmentNameFieldId}, Location FID ${lessonState.departmentLocationFieldId}.`;

          output.textContent = JSON.stringify(
            {
              quickbaseAutomaticFieldsAlreadyPresent: [
                { id: 1, label: "Date Created" },
                { id: 2, label: "Date Modified" },
                { id: 3, label: "Record ID#" },
                { id: 4, label: "Record Owner" },
                { id: 5, label: "Last Modified By" },
              ],
              fieldsCreatedByOurApiRequests: fieldResults,
              capturedLessonState: lessonState,
            },
            null,
            2,
          );
          await inspectLessonSchema();
        } catch (error) {
          console.error(error);

          status.classList.remove("success");
          status.textContent = `Create fields failed: ${error.message}`;
          output.textContent = JSON.stringify(
            {
              error: error.message,
              lessonState,
            },
            null,
            2,
          );

          button.disabled = false;
        }
      }

      async function discoverPeopleTable() {
        if (!APP_DBID) {
          throw new Error("Confirm the Quickbase app context first.");
        }

        const temporaryToken = await getTemporaryToken(APP_DBID);
        const endpoint = `https://api.quickbase.com/v1/tables?appId=${APP_DBID}`;

        const response = await fetch(endpoint, {
          method: "GET",
          headers: {
            "QB-Realm-Hostname": REALM,
            Authorization: `QB-TEMP-TOKEN ${temporaryToken}`,
            "Content-Type": "application/json",
          },
        });

        const data = await response.json();

        if (!response.ok) {
          throw new Error(`HTTP ${response.status}: ${JSON.stringify(data)}`);
        }

        const peopleTable = data.find(
          (table) => table.name.trim().toLowerCase() === "people",
        );

        if (!peopleTable) {
          throw new Error('Could not find a table named "People".');
        }

        lessonState.peopleTableDbid = peopleTable.id;

        return {
          endpoint,
          matchedTable: peopleTable,
        };
      }

              label: "Related Department",

            lookupFieldIds: [

          const response = await fetch(endpoint, {
            method: "POST",

              "Content-Type": "application/json",

                  method: "GET",

                method: "POST",
