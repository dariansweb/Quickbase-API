<script>
      const REALM = window.location.hostname.includes("quickbase.com")
        ? window.location.hostname
        : "YOUR_REALM.quickbase.com";

      function getAppDbid() {
        const match = window.location.pathname.match(/\/db\/([^/]+)/);

        if (!match) {
          throw new Error(
            "Could not determine the App DBID from the Quickbase URL.",
          );
        }

        return match[1];
      }

      const APP_TOKEN = "";
      const APP_DBID = getAppDbid();

      /*
       * Cache the app's table metadata so relationship DBIDs can
       * also be shown with human-readable table names.
       */
      let appTables = [];
      /*
       * ============================================================
       * TEMPORARY REST AUTHORIZATION
       * ============================================================
       *
       * This is the same temporary-token pattern used in our
       * previous REST lessons.
       */
      async function getTemporaryToken(dbid) {
        const headers = {
          "QB-Realm-Hostname": REALM,
        };

        if (APP_TOKEN.trim()) {
          headers["QB-App-Token"] = APP_TOKEN;
        }

        const response = await fetch(
          `https://api.quickbase.com/v1/auth/temporary/${dbid}`,
          {
            method: "GET",
            credentials: "include",
            headers,
          },
        );

        const data = await response.json();

        if (!response.ok) {
          throw new Error(`HTTP ${response.status}: ${JSON.stringify(data)}`);
        }

        if (!data.temporaryAuthorization) {
          throw new Error("Quickbase did not return temporaryAuthorization.");
        }

        return data.temporaryAuthorization;
      }

      async function copyOutputToClipboard() {
        const output = document.getElementById("output");

        const status = document.getElementById("status");

        const text = output.textContent.trim();

        if (!text || text === "No response yet." || text === "Loading...") {
          status.textContent = "There is no JSON response to copy yet.";

          return;
        }

        try {
          await navigator.clipboard.writeText(text);

          status.classList.remove("error");

          status.textContent = "JSON response copied to clipboard.";
        } catch (error) {
          console.error(error);

          status.classList.add("error");

          status.textContent = "Unable to copy JSON to the clipboard.";
        }
      }

      function getSelectedTable() {
        const tableDbid = getSelectedTableDbid();

        return appTables.find((table) => table.id === tableDbid) || null;
      }

      function getTableNameByDbid(dbid) {
        const table = appTables.find((item) => item.id === dbid);

        return table ? table.name : "Unknown table";
      }

      function updateContextDisplay() {
        document.getElementById("appDbidDisplay").textContent = APP_DBID;

        const tableSelect = document.getElementById("tableSelect");
        const selectedOption = tableSelect.options[tableSelect.selectedIndex];

        document.getElementById("selectedTableDisplay").textContent =
          selectedOption && selectedOption.value
            ? selectedOption.textContent
            : "None";
      }

      function addRelationshipTableNames(data) {
        return {
          ...data,
          relationships: (data.relationships || []).map((relationship) => ({
            ...relationship,
            parentTableName: getTableNameByDbid(relationship.parentTableId),
            childTableName: getTableNameByDbid(relationship.childTableId),
          })),
        };
      }

      /*
       * ============================================================
       * GET RELATIONSHIPS
       * ============================================================
       *
       * Ask Quickbase to describe the relationships for the
       * selected child table.
       *
       * For now we do NOT transform the response.
       * We want to see the raw JSON exactly as Quickbase returns it.
       */
      async function getRelationships() {
        const status = document.getElementById("status");

        const output = document.getElementById("output");

        status.classList.remove("error");

        status.textContent = "Requesting relationship schema from Quickbase...";

        output.textContent = "Loading...";

        try {
          const tableDbid = getSelectedTableDbid();
          const temporaryToken = await getTemporaryToken(tableDbid);
          const endpoint = `https://api.quickbase.com/v1/tables/${tableDbid}/relationships`;

          console.log("Relationship endpoint:", endpoint);

          const response = await fetch(endpoint, {
            method: "GET",

            headers: {
              "QB-Realm-Hostname": REALM,

              Authorization: `QB-TEMP-TOKEN ${temporaryToken}`,

              "Content-Type": "application/json",
            },
          });

          const data = await response.json();

          console.log("Relationship response:", data);

          if (!response.ok) {
            throw new Error(`HTTP ${response.status}: ${JSON.stringify(data)}`);
          }

          const resolvedData = addRelationshipTableNames(data);

          status.textContent = `HTTP ${response.status} — relationship schema received.`;

          output.textContent = JSON.stringify(
            {
              selectedTable: getSelectedTable(),
              relationshipSchema: resolvedData,
            },
            null,
            2,
          );
        } catch (error) {
          console.error(error);

          status.classList.add("error");

          status.textContent = `Request failed: ${error.message}`;

          output.textContent = "No relationship data returned.";
        }
      }

      async function getTable() {
        const status = document.getElementById("status");

        const output = document.getElementById("output");

        status.classList.remove("error");

        status.textContent = "Requesting table schema from Quickbase...";

        output.textContent = "Loading...";

        try {
          const tableDbid = getSelectedTableDbid();
          const temporaryToken = await getTemporaryToken(APP_DBID);

          const endpoint = `https://api.quickbase.com/v1/tables/${tableDbid}?appId=${APP_DBID}`;
          console.log("Table endpoint:", endpoint);

          const response = await fetch(endpoint, {
            method: "GET",

            headers: {
              "QB-Realm-Hostname": REALM,

              Authorization: `QB-TEMP-TOKEN ${temporaryToken}`,

              "Content-Type": "application/json",
            },
          });

          const data = await response.json();

          console.log("Table response:", data);

          if (!response.ok) {
            throw new Error(`HTTP ${response.status}: ${JSON.stringify(data)}`);
          }

          status.textContent = `HTTP ${response.status} — table schema received.`;

          output.textContent = JSON.stringify(data, null, 2);
        } catch (error) {
          console.error(error);

          status.classList.add("error");

          status.textContent = `Request failed: ${error.message}`;

          output.textContent = "No table data returned.";
        }
      }

      async function getFields() {
        const status = document.getElementById("status");

        const output = document.getElementById("output");

        status.classList.remove("error");

        status.textContent = "Requesting field schema from Quickbase...";

        output.textContent = "Loading...";

        try {
          const tableDbid = getSelectedTableDbid();
          const temporaryToken = await getTemporaryToken(tableDbid);
          const endpoint = `https://api.quickbase.com/v1/fields?tableId=${tableDbid}`;

          console.log("Fields endpoint:", endpoint);

          const response = await fetch(endpoint, {
            method: "GET",

            headers: {
              "QB-Realm-Hostname": REALM,

              Authorization: `QB-TEMP-TOKEN ${temporaryToken}`,

              "Content-Type": "application/json",
            },
          });

          const data = await response.json();

          console.log("Fields response:", data);

          if (!response.ok) {
            throw new Error(`HTTP ${response.status}: ${JSON.stringify(data)}`);
          }

          status.textContent = `HTTP ${response.status} — field schema received.`;

          output.textContent = JSON.stringify(data, null, 2);
        } catch (error) {
          console.error(error);

          status.classList.add("error");

          status.textContent = `Request failed: ${error.message}`;

          output.textContent = "No field data returned.";
        }
      }

      /*
       * ============================================================
       * GET FULL SCHEMA SNAPSHOT
       * ============================================================
       *
       * Collect the selected table's table metadata, fields, and
       * relationships into one JSON document.
       *
       * Relationship DBIDs are also resolved to table names.
       */
      async function getFullSchema() {
        const status = document.getElementById("status");
        const output = document.getElementById("output");

        status.classList.remove("error");
        status.textContent =
          "Building complete schema snapshot from Quickbase...";
        output.textContent = "Loading...";

        try {
          const tableDbid = getSelectedTableDbid();

          const appToken = await getTemporaryToken(APP_DBID);
          const tableToken = await getTemporaryToken(tableDbid);

          const tableEndpoint = `https://api.quickbase.com/v1/tables/${tableDbid}?appId=${APP_DBID}`;
          const fieldsEndpoint = `https://api.quickbase.com/v1/fields?tableId=${tableDbid}`;
          const relationshipsEndpoint = `https://api.quickbase.com/v1/tables/${tableDbid}/relationships`;

          const commonHeaders = {
            "QB-Realm-Hostname": REALM,
            "Content-Type": "application/json",
          };

          const [tableResponse, fieldsResponse, relationshipsResponse] =
            await Promise.all([
              fetch(tableEndpoint, {
                method: "GET",
                headers: {
                  ...commonHeaders,
                  Authorization: `QB-TEMP-TOKEN ${appToken}`,
                },
              }),
              fetch(fieldsEndpoint, {
                method: "GET",
                headers: {
                  ...commonHeaders,
                  Authorization: `QB-TEMP-TOKEN ${tableToken}`,
                },
              }),
              fetch(relationshipsEndpoint, {
                method: "GET",
                headers: {
                  ...commonHeaders,
                  Authorization: `QB-TEMP-TOKEN ${tableToken}`,
                },
              }),
            ]);

          const tableData = await tableResponse.json();
          const fieldsData = await fieldsResponse.json();
          const relationshipsData = await relationshipsResponse.json();

          if (!tableResponse.ok) {
            throw new Error(
              `Table HTTP ${tableResponse.status}: ${JSON.stringify(tableData)}`,
            );
          }

          if (!fieldsResponse.ok) {
            throw new Error(
              `Fields HTTP ${fieldsResponse.status}: ${JSON.stringify(fieldsData)}`,
            );
          }

          if (!relationshipsResponse.ok) {
            throw new Error(
              `Relationships HTTP ${relationshipsResponse.status}: ${JSON.stringify(relationshipsData)}`,
            );
          }

          const snapshot = {
            app: {
              realm: REALM,
              appDbid: APP_DBID,
            },
            selectedTable: {
              name: tableData.name,
              dbid: tableData.id,
            },
            table: tableData,
            fields: fieldsData,
            relationships: addRelationshipTableNames(relationshipsData),
          };

          status.textContent = `Full schema snapshot received for ${tableData.name} — ${tableData.id}.`;

          output.textContent = JSON.stringify(snapshot, null, 2);
        } catch (error) {
          console.error(error);

          status.classList.add("error");
          status.textContent = `Request failed: ${error.message}`;
          output.textContent = "No full schema snapshot returned.";
        }
      }

      async function getTables() {
        const status = document.getElementById("status");

        const tableSelect = document.getElementById("tableSelect");

        status.classList.remove("error");

        status.textContent = "Loading tables from Quickbase...";

        try {
          const temporaryToken = await getTemporaryToken(APP_DBID);

          const endpoint = `https://api.quickbase.com/v1/tables?appId=${APP_DBID}`;

          const response = await fetch(endpoint, {
            method: "GET",

            headers: {
              "QB-Realm-Hostname": REALM,

              Authorization: `QB-TEMP-TOKEN ${temporaryToken}`,

              "Content-Type": "application/json",
            },
          });

          const data = await response.json();

          if (!response.ok) {
            throw new Error(`HTTP ${response.status}: ${JSON.stringify(data)}`);
          }

          appTables = data;

          /*
           * Remove the temporary "Loading tables..." option.
           */
          tableSelect.replaceChildren();

          /*
           * Add one <option> for every table Quickbase returned.
           */
          for (const table of data) {
            const option = document.createElement("option");

            option.value = table.id;

            option.textContent = `${table.name} — ${table.id}`;

            tableSelect.appendChild(option);
          }

          status.textContent = `${data.length} Quickbase table(s) loaded.`;

          updateContextDisplay();
        } catch (error) {
          console.error(error);

          status.classList.add("error");

          status.textContent = `Unable to load tables: ${error.message}`;

          tableSelect.replaceChildren();

          const option = document.createElement("option");

          option.value = "";
          option.textContent = "Unable to load tables";

          tableSelect.appendChild(option);
        }
      }

        const tableSelect = document.getElementById("tableSelect");

        }

        return tableDbid;
